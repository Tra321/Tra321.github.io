本次项目使用Docker Compose一键部署前后端分离项目，前端使用Vue，后端使用Spring Boot，数据库使用MySQL，使用Nginx作为反向代理。以CentOS7.9为例

[Docker和Docker Compose安装教程](https://tra321.github.io/2025/02/16/2025-02-16-Docker%E4%BF%9D%E5%A7%86%E7%BA%A7%E5%AE%89%E8%A3%85%E6%95%99%E7%A8%8B/)

# 二、 准备工作

## 1、新建工作目录

在/home目录下新建目录sky，作为工作目录

```bash
[root@localhost home]# mkdir sky
```

## 2、工作目录结构

> 其中conf目录存放redis.conf和nginx.conf配置  
> 其中db目录存放sky数据库脚本  
> 数据库脚本头部需要添加SET NAMES ‘utf8’;（防止乱码）  
> 数据库mysql地址需要修改成sky-mysql  
> 缓存redis地址需要修改成sky-redis  
> 其中html\\dist目录存放打包好的静态页面文件  
> 其中jar目录存放打包好的jar应用文件

```bash
[root@localhost sky]# tree

├── conf
│   ├── nginx.conf
│   └── redis.conf
├── db

├── mysql-dockerfile

```

### nginx.conf文件内容

```bash
worker_processes  1;

events {
    worker_connections  1024;
}

http {
    include       mime.types;
    default_type  application/octet-stream;

    sendfile        on;

    keepalive_timeout  65;
	
	map $http_upgrade $connection_upgrade{
		default upgrade;
		'' close;
	}

	upstream webservers{
	  server sky-server:8080;
	}

    server {
        listen       80;
        server_name  localhost;

        location / {
            root   /usr/share/nginx/html;
            try_files $uri $uri/ /index.html;
            index  index.html index.htm;
        }

        error_page   500 502 503 504  /50x.html;
        location = /50x.html {
            root   html;
        }

        # 反向代理,处理管理端发送的请求
        location /api/ {
			proxy_pass   http://sky-server:8080/admin/;
        }
		
		# 反向代理,处理用户端发送的请求
        location /user/ {
            proxy_pass   http://webservers/user/;
        }
		
		# WebSocket
		location /ws/ {
            proxy_pass   http://webservers/ws/;
			proxy_http_version 1.1;
			proxy_read_timeout 3600s;
			proxy_set_header Upgrade $http_upgrade;
			proxy_set_header Connection "$connection_upgrade";
        }
    }
}
```

### 编写mysql-dockerfile

mysql-dockerfile 文件内容如下：

```bash
# 基础镜像
FROM mysql:8.0

# author
LABEL maintainer="sky"
 
# 执行sql脚本
ADD ./db/*.sql /docker-entrypoint-initdb.d/
```

### 编写redis-dockerfile

redis-dockerfile 文件内容如下：

```bash
# 基础镜像
FROM nginx

# author
LABEL maintainer="sky"
 
# 挂载目录
VOLUME /home/sky/projects/sky
# 创建目录
RUN mkdir -p /home/sky/projects/sky
# 指定路径
WORKDIR /home/sky/projects/sky
# 复制conf文件到路径
COPY ./conf/nginx.conf /etc/nginx/nginx.conf
# 复制html文件到路径
COPY ./html/dist /home/sky/projects/sky
```

### 编写sky-dockerfile

sky-dockerfile 文件内容如下：

```bash
# 基础镜像
FROM openjdk:17.0.2
# author
LABEL maintainer="sky"
 
# 挂载目录
VOLUME /home/sky
# 创建目录
RUN mkdir -p /home/sky
# 指定路径
WORKDIR /home/sky
# 复制jar文件到路径
COPY ./jar/*.jar /home/sky/sky-server-1.0.jar
# 启动应用
ENTRYPOINT ["java","-jar","sky-server-1.0.jar"]
```

### 编写docker-compose.yml

docker-compose.yml 文件内容如下：

```bash
version: '3'
services:
  sky-nginx:
    image: nginx:1.20.1
    container_name: sky-nginx
    build:
      context: .
      dockerfile: nginx-dockerfile
    ports:
      - "80:80"
    volumes:
      - ./html/dist/sky:/usr/share/nginx/html
      - ./conf/nginx.conf:/etc/nginx/nginx.conf
      - ./nginx/logs:/var/log/nginx
      - ./nginx/conf.d:/etc/nginx/conf.d
    depends_on:
      - sky-server
    links:
      - sky-server

  sky-server:
    image: openjdk:17.0.2
    container_name: sky-server
    build:
      context: .
      dockerfile: sky-dockerfile
    ports:
      - "8080:8080"
    volumes:
      - ./sky/logs:/home/sky/logs
      - ./sky/uploadPath:/home/sky/uploadPath
    depends_on:
      - sky-redis
      - sky-mysql
    links:
      - sky-mysql
      - sky-redis

  sky-redis:
    image: redis:3.2.1
    container_name: sky-redis
    build:
      context: .
      dockerfile: redis-dockerfile
    ports:
      - "6379:6379"
    volumes:
      - ./conf/redis.conf:/home/sky/redis/redis.conf
      - ./redis/data:/data
    command: redis-server /home/sky/redis/redis.conf

  sky-mysql:
    image: mysql:8.0
    container_name: sky-mysql
    build:
      context: .
      dockerfile: mysql-dockerfile
    environment:
      MYSQL_ROOT_PASSWORD: Mysql135!
      MYSQL_DATABASE: 'sky_take_out'
    ports:
      - "3306:3306"
    volumes:
      - ./mysql/conf:/etc/mysql/conf.d
      - ./mysql/logs:/logs
      - ./mysql/data:/var/lib/mysql
    command: [
          'mysqld',
          '--innodb-buffer-pool-size=80M',
          '--character-set-server=utf8mb4',
          '--collation-server=utf8mb4_unicode_ci',
          '--default-time-zone=+8:00',
          '--lower-case-table-names=1'
        ]
```

## 3、构建并启动

### 构建docker服务

```bash
[root@localhost sky]# docker-compose build
```

### 启动docker容器

```bash
[root@localhost sky]# docker-compose up -d
```

### 查看容器

```bash
[root@localhost sky]# docker ps
```

### 浏览器访问测试

可以正常显示页面，表名环境搭建成功。
